   138→/* ------------------------------------------------------- terminal scan */
   139→
   140→/**
   141→ * Progressive `wanyrix doctor` terminal. Remounted (via key={runId}) to replay:
   142→ * initial state carries the reset, and setState is only ever called from
   143→ * timer callbacks — never synchronously inside the effect body.
   144→ */
   145→function ScanTerminal({
   146→  report,
   147→  onDone,
   148→}: {
   149→  report: DoctorReport
   150→  onDone: (done: boolean, durationMs: number) => void
   151→}) {
   152→  const [revealed, setRevealed] = useState(1) // the $ command line is visible immediately
   153→  const [done, setDone] = useState(false)
   154→  const total = 1 + report.phases.length + 2 // cmd + phases + 2 summary lines
   155→
   156→  /* keep the latest callback without restarting the reveal timers (issue #25 pattern) */
   157→  const onDoneRef = useRef(onDone)
   158→  useEffect(() => {
   159→    onDoneRef.current = onDone
   160→  }, [onDone])
   161→  const mountedAt = useRef(0)
   162→
   163→  useEffect(() => {
   164→    mountedAt.current = performance.now()
   165→    let count = 1
   166→    let doneTimer: ReturnType<typeof setTimeout> | undefined
   167→    const iv = setInterval(() => {
   168→      count += 1
   169→      setRevealed(count)
   170→      if (count >= total) {
   171→        clearInterval(iv)
   172→        doneTimer = setTimeout(() => {
   173→          setDone(true)
   174→          onDoneRef.current(true, performance.now() - mountedAt.current)
   175→        }, 480)
   176→      }
   177→    }, 380)
   178→    return () => {
   179→      clearInterval(iv)
   180→      if (doneTimer) clearTimeout(doneTimer)
   181→    }
   182→  }, [total])
   183→
   184→  const lines = [
   185→    { text: 'wanyrix doctor --profile dev', tone: 'cmd' as const },
   186→    ...report.phases.map((p) => ({ text: `  ✓ ${p.label} — ${p.detail}`, tone: 'ok' as const })),
   187→    {
   188→      text: `Wanyrix found ${report.findings.length} engineering bottlenecks.`,
   189→      tone: 'accent' as const,
   190→    },
   191→    {
   192→      text: `Estimated opportunity: dev ${report.summary.developerBuild} · CI ${report.summary.ciBuild} · disk ${report.summary.diskUsage} (estimated)`,
   193→      tone: 'warn' as const,
   194→    },
   195→  ]
   196→
   197→  return (
   198→    <Terminal
   199→      title="wanyrix — zsh"
   200→      lines={lines.slice(0, revealed)}
   201→      step={0}
   202→      running={!done}
   203→    />
   204→  )
   205→}

   209→function FindingCard({

   355→export default function DoctorView({ onNavigate }: ViewProps) {
   356→  const { data: report, isLoading, isError, error, refetch } = useDoctor()
   357→  const { toast } = useToast()
   358→  const activeWs = useWorkspaceStore((s) => s.active)
   359→  const scanTick = useScanStore((s) => s.scanTick)
   360→  const lastTrigger = useScanStore((s) => s.lastTrigger)
   361→  const bumpScan = useScanStore((s) => s.bumpScan)
   362→  const addScanEntry = useScanStore((s) => s.addEntry)
   363→
   364→  const [mode, setMode] = useState<'human' | 'json'>('human')
   365→  const [selected, setSelected] = useState<Finding | null>(null)
   366→
   367→  /* every run goes through the global scan event (issue #37): runId = scanTick.
   368→     The doctor button bumps it with trigger 'manual'; the topbar / ⌘K bump it
   369→     with their own triggers. ScanTerminal remounts on every runId change. */
   370→  const runId = scanTick
   371→  const [doneRun, setDoneRun] = useState(-1)
   372→  const scanDone = doneRun === runId
   373→  const mountedAtTick = useRef(scanTick)
   374→
   375→  /* findings grouped by section, in curated data order */
   376→  const groups = useMemo(() => {
   377→    if (!report) return []
   378→    const bySection = new Map<FindingSection, Finding[]>()
   379→    for (const f of report.findings) {
   380→      const list = bySection.get(f.section) ?? []
   381→      list.push(f)
   382→      bySection.set(f.section, list)
   383→    }
   384→    const ordered: FindingSection[] = [
   385→      ...SECTION_ORDER.filter((s) => bySection.has(s)),
   386→      ...[...bySection.keys()].filter((s) => !SECTION_ORDER.includes(s)),
   387→    ]
   388→    return ordered.map((section) => ({ section, findings: bySection.get(section)! }))
   389→  }, [report])
   390→
   391→  /* recomputed on every runId replay (ScanTerminal builds its own copy) */
   392→  const json = report ? JSON.stringify(report, null, 2) : ''
   393→
   394→  /* scan completion → record a history entry (issue #37); figures from the payload.
   395→     Trigger label: 'manual' for the auto-run on view mount, otherwise whatever
   396→     control bumped the scan event that produced this run. */
   397→  const handleScanDone = (done: boolean, durationMs: number) => {
   398→    if (!done) return
   399→    setDoneRun(runId)
   400→    if (report) {
   401→      addScanEntry(activeWs, {
   402→        id: `scan-${Date.now()}`,
   403→        workspace: activeWs,
   404→        at: Date.now(),
   405→        durationMs,
   406→        findings: report.findings.length,
   407→        critical: report.findings.filter((f) => f.severity === 'critical').length,
   408→        warning: report.findings.filter((f) => f.severity === 'warning').length,
   409→        info: report.findings.filter((f) => f.severity === 'info').length,
   410→        buildTime: report.buildTime,
   411→        estimatedFrom: report.estimatedRange[0],
   412→        estimatedTo: report.estimatedRange[1],
   413→        trigger: scanTick === mountedAtTick.current ? 'manual' : lastTrigger,
   414→      })
   415→    }
   416→  }
   417→
   418→  const copyJson = () => {
   419→    if (!navigator.clipboard) {
   420→      toast({ title: 'Copy failed', description: 'Clipboard is not available in this context.' })
   421→      return
   422→    }
   423→    navigator.clipboard.writeText(json).then(
   424→      () =>
   425→        toast({
   426→          title: 'Copied — schema versioned (Gate 28)',
   427→          description: 'wanyrix doctor --json output is on your clipboard.',
   428→        }),
   429→      () => toast({ title: 'Copy failed', description: 'Clipboard permission denied.' }),
   430→    )
   431→  }
   432→
   433→  /* human-readable markdown report — same facts as --json, Gate 11 equivalence */
   434→  const markdown = report
   435→    ? [
   436→        `# Wanyrix Build Analysis — ${report.workspace}`,
   437→        '',
   438→        `Development build: **${report.buildTime}s** (measured · cargo build --timings)`,
   439→        `Estimated after fixes: **${report.estimatedRange[0]}–${report.estimatedRange[1]}s** (estimated, confidence ${report.confidence}%)`,
   440→        '',
   441→        '## Critical path',
   442→        ...report.criticalPath.map((s) => `- ${s.name}: ${s.seconds}s`),
   443→        '',
   444→        '## Findings',
   445→        ...(report.findings ?? []).flatMap((f) => [
   446→          `### ${f.id} · ${f.severity} · ${f.title} [${f.measurementStatus}]`,
   447→          f.description,
   448→          '',
   449→          '**Evidence**',
   450→          ...f.evidence.map((e) => `- ${e.label}: ${e.value} _(src: ${e.source})_`),
   451→          `- Impact: ${f.impact}`,
   452→          `- Recommendation: ${f.recommendation}`,
   453→          `- Verification: ${f.verificationPath}`,
   454→          '',
   455→        ]),
   456→        '---',
   457→        '_Estimated values — verified only via wanyrix experiment (Gate 21)._',
   458→      ].join('\n')
   459→    : ''
   460→
   461→  const copyMarkdown = () => {
   462→    if (!navigator.clipboard) {
   463→      toast({ title: 'Copy failed', description: 'Clipboard is not available in this context.' })
   464→      return
   465→    }
   466→    navigator.clipboard.writeText(markdown).then(
   467→      () =>
   468→        toast({
   469→          title: 'Report copied as markdown',
   470→          description: 'Human/JSON outputs expose equivalent underlying facts (Gate 11).',
   471→        }),
   472→      () => toast({ title: 'Copy failed', description: 'Clipboard permission denied.' }),
   473→    )
   474→  }
   475→
   476→  if (isLoading) return <LoadingSkeleton />
   477→  if (isError || !report)
   478→    return (
   479→      <ErrorState
   480→        message={isError ? (error as Error).message : 'empty payload'}
   481→        onRetry={() => refetch()}
   482→      />
   483→    )
   484→
   485→  return (
   486→    <div className="space-y-5">
   487→      {/* ------------------------------------------------ 1) header */}
   488→      <SectionHeading
   489→        eyebrow="Build Intelligence"
   490→        title="wanyrix doctor"
   491→        description={`${report.workspace} · ${report.profile} · ${report.toolchain}`}
   492→        actions={
   493→          <>
   494→            <ToggleGroup
   495→              type="single"
   496→              size="sm"
   497→              variant="outline"
   498→              value={mode}
   499→              onValueChange={(v) => v && setMode(v as 'human' | 'json')}
   500→              aria-label="Output mode"
   501→            >
   502→              <ToggleGroupItem value="human">Human</ToggleGroupItem>
   503→              <ToggleGroupItem value="json" className="font-mono">
   504→                --json
   505→              </ToggleGroupItem>
   506→            </ToggleGroup>
   507→            <Button
   508→              size="sm"
   509→              variant="outline"
   510→              className="gap-2"
   511→              onClick={copyMarkdown}
   512→              disabled={!scanDone}
   513→              title="Copy the human report as markdown (Gate 11)"
   514→            >
   515→              <ClipboardCopy className="size-4" />
   516→              <span className="hidden sm:inline">Copy report (md)</span>
   517→            </Button>
   518→            <Button
   519→              size="sm"
   520→              onClick={() => {
   521→                setMode('human') // a fresh run is watched in the terminal, not in JSON
   522→                bumpScan('manual') // runId = scanTick → ScanTerminal remounts and replays
   523→              }}
   524→              disabled={!scanDone}
   525→              className="gap-2"
   526→            >
   527→              <RefreshCw className={`size-4 ${!scanDone ? 'animate-spin' : ''}`} />
   528→              Run wanyrix doctor
   529→            </Button>
   530→          </>
   531→        }
   532→      />
   533→
   534→      {/* ------------------------------------------------ 2) terminal / json */}
   535→      {/* the terminal always runs (so a scan is recorded even while JSON mode is
   536→          on); it is simply hidden while the JSON report is being viewed */}
   537→      <div className={mode === 'json' ? 'hidden' : undefined}>
   538→        <ScanTerminal key={runId} report={report} onDone={handleScanDone} />
   539→      </div>
   540→      {mode === 'json' && (
   541→        <div className="space-y-2">
   542→          <Panel
   543→            title={<span className="font-mono">wanyrix doctor --json</span>}
