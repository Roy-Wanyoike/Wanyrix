    71→function ImpactTile({
    72→  label,
    73→  value,
    74→  decimals = 0,
    75→  prefix = '',
    76→  suffix = '',
    77→  cost = false,
    78→}: {
    79→  label: string
    80→  value: number
    81→  decimals?: number
    82→  prefix?: string
    83→  suffix?: string
    84→  cost?: boolean
    85→}) {
    86→  // negative deltas are genuine improvements (e.g. de-dup wins in upgrade-dep)
    87→  const improvement = value < 0
    88→  return (
    89→    <div
    90→      className={cn(
    91→        'rounded-lg border px-3 py-2.5 transition-shadow',
    92→        improvement
    93→          ? 'border-emerald-500/30 bg-emerald-500/[0.07] shadow-[0_0_16px_-8px_oklch(0.72_0.17_160/60%)]'
    94→          : 'border-border/70 bg-muted/20',
    95→      )}
    96→    >
    97→      <p className="text-[10px] uppercase tracking-wide text-muted-foreground">{label}</p>
    98→      <p
    99→        className={cn(
   100→          'mt-1 font-mono text-lg font-semibold',
   101→          improvement ? 'text-emerald-300' : cost ? 'text-red-300' : 'text-foreground',
   102→        )}
   103→      >
   104→        {improvement && <TrendingDown className="mr-1 inline size-3.5 align-baseline" aria-hidden />}
   105→        <CountUp
   106→          value={Math.abs(value)}
   107→          decimals={decimals}
   108→          prefix={improvement ? '−' : prefix}
   109→          suffix={suffix}
   110→        />
   111→      </p>
   112→    </div>
   113→  )
   114→}

   182→function AddDepReport({ query }: { query: QueryLike<ImpactPayload> }) {
   183→  const [pending, setPending] = useState<string | null>(null)
   184→  const { toast } = useToast()
   185→  const activeWs = useWorkspaceStore((s) => s.active)
   186→  const enqueueDiff = useDiffQueueStore((s) => s.enqueue)
   187→  const d = query.data && query.data.kind === 'add-dep' ? (query.data as AddDepImpact) : undefined
   188→
   189→  if (query.isPending || (query.data === undefined && !query.isError)) {
   190→    return (
   191→      <Panel title="Dependency impact">
   192→        <TilesSkeleton />
   193→      </Panel>
   194→    )
   195→  }
   196→
   197→  if (!d) {
   198→    return (
   199→      <Panel title="Dependency impact">
   200→        <QueryError message={query.error?.message ?? 'unknown error'} onRetry={() => query.refetch()} />
   201→      </Panel>
   202→    )
   203→  }
   204→
   205→  const tiles: {
   206→    label: string
   207→    value: number
   208→    decimals?: number
   209→    prefix?: string
   210→    suffix?: string
   211→    cost?: boolean
   212→  }[] = [
   213→    { label: 'crates added', value: d.cratesAdded },
   214→    { label: 'proc-macros added', value: d.procMacrosAdded },
   215→    { label: 'target size', value: d.targetMB, decimals: 1, suffix: ' MB' },
   216→    { label: 'transitive deps', value: d.transitiveDeps },
   217→    { label: 'clean build', value: d.cleanDelta, decimals: 1, prefix: '+', suffix: 's', cost: true },
   218→    { label: 'incremental', value: d.incrementalDelta, decimals: 1, prefix: '+', suffix: 's', cost: true },
   219→    { label: 'CI', value: d.ciDelta, prefix: '+', suffix: 's', cost: true },
   220→  ]
   221→
   222→  return (
   223→    <Panel
   224→      title={`Dependency impact — ${d.crate} ${d.version}`}
   225→      subtitle="what enters your tree if you add this crate"
   226→      actions={<MeasurementBadge status={d.measurementStatus} />}
   227→    >
   228→      <div className="space-y-5">
   229→        <div className="grid grid-cols-2 gap-2.5 md:grid-cols-4">
   230→          {tiles.map((t) => (
   231→            <ImpactTile key={t.label} {...t} />
   232→          ))}
   233→        </div>
   234→
   235→        <section>
   236→          <SectionLabel>Cascading rebuild chain</SectionLabel>
   237→          <div className="mt-3">
   238→            <ChainStepper items={d.chain} />
   239→          </div>
   240→        </section>
   241→
   242→        <section>
   243→          <SectionLabel>Security &amp; supply chain</SectionLabel>
   244→          <ul className="mt-2.5 space-y-1.5">
   245→            {d.security.map((s, i) => (
   246→              <li key={i} className="flex items-start gap-2 text-xs">
   247→                {s.level === 'warn' ? (
   248→                  <ShieldAlert className="mt-0.5 size-3.5 shrink-0 text-amber-400" />
   249→                ) : (
   250→                  <ShieldCheck className="mt-0.5 size-3.5 shrink-0 text-teal-300" />
   251→                )}
   252→                <span className={s.level === 'warn' ? 'text-amber-200/90' : 'text-muted-foreground'}>
   253→                  {s.label}
   254→                </span>
   255→              </li>
   256→            ))}
   257→          </ul>
   258→        </section>
   259→
   260→        <section>
   261→          <SectionLabel>Suggestions</SectionLabel>
   262→          <ol className="mt-2.5 space-y-2">
   263→            {d.suggestions.map((s, i) => (
   264→              <li
   265→                key={i}
   266→                className="flex items-center justify-between gap-3 rounded-lg border border-border/60 bg-muted/20 px-3 py-2"
   267→              >
   268→                <p className="text-xs leading-snug">
   269→                  <span className="mr-2 font-mono text-muted-foreground">{String(i + 1).padStart(2, '0')}</span>
   270→                  {s}
   271→                </p>
   272→                <Button
   273→                  size="sm"
   274→                  variant="outline"
   275→                  className="h-7 shrink-0 text-[11px]"
   276→                  onClick={() => setPending(s)}
   277→                >
   278→                  Apply suggestion
   279→                </Button>
   280→              </li>
   281→            ))}
   282→          </ol>
   283→        </section>
   284→      </div>
   285→
   286→      <Dialog open={pending !== null} onOpenChange={(o) => !o && setPending(null)}>
   287→        <DialogContent className="max-w-md">
   288→          <DialogHeader>
   289→            <DialogTitle>Queue a reviewable diff?</DialogTitle>
   290→            <DialogDescription>
   291→              Ferrix never modifies your repository silently. A reviewable diff will be proposed (Gate 19).
   292→            </DialogDescription>
   293→          </DialogHeader>
   294→          <div className="rounded-lg border border-border/70 bg-muted/30 px-3 py-2.5 font-mono text-[11px] leading-relaxed text-foreground/90">
   295→            {pending}
   296→          </div>
   297→          <div className="flex justify-end gap-2">
   298→            <Button size="sm" variant="ghost" onClick={() => setPending(null)}>
   299→              Cancel
   300→            </Button>
   301→            <Button
   302→              size="sm"
   303→              onClick={() => {
   304→                if (!pending) return
   305→                enqueueDiff({
   306→                  id: `diff-${Date.now()}`,
   307→                  workspace: activeWs,
   308→                  at: Date.now(),
   309→                  source: 'simulator:add-dep',
   310→                  kind: 'config',
   311→                  title: `Dependency change — ${d.crate} ${d.version}`,
   312→                  target: `${d.crate} ${d.version} · Cargo.toml`,
   313→                  suggestion: pending,
   314→                  estimate: `+${d.cleanDelta}s clean · +${d.ciDelta}s CI per build`,
   315→                  status: 'pending',
   316→                })
   317→                toast({
   318→                  title: 'Diff queued for review',
   319→                  description: 'Open Pending diffs in the topbar to inspect or export it (Gate 19).',
   320→                })
   321→                setPending(null)
   322→              }}
   323→            >
   324→              Queue as diff
   325→            </Button>
   326→          </div>
   327→        </DialogContent>
   328→      </Dialog>
   329→    </Panel>
   330→  )
   331→}

   470→      <Panel title="Architecture proposal">
