    31→const BAND_Y = { bin: 90, lib: 320, external: 550 } as const
    32→const X_MIN = 60
    33→const X_MAX = 900
    34→
    35→interface Positioned extends GraphNode {
    36→  x: number
    37→  y: number
    38→  r: number
    39→}
    40→
    41→/**
    42→ * Band layout: workspace bins (y 90) → workspace libs (y 320) → externals &
    43→ * proc-macros (y 550). Band members spread evenly across X_MIN..X_MAX in the
    44→ * exact order they appear in the payload's nodes array.
    45→ */
    46→function layoutGraph(nodes: GraphNode[]): Map<string, Positioned> {
    47→  const map = new Map<string, Positioned>()
    48→  for (const band of ['bin', 'lib', 'external'] as const) {
    49→    const members = nodes.filter((n) => n.band === band)
    50→    members.forEach((n, i) => {
    51→      const x =
    52→        members.length === 1 ? (X_MIN + X_MAX) / 2 : X_MIN + (i * (X_MAX - X_MIN)) / (members.length - 1)
    53→      const r = Math.min(22, Math.max(9, 7 + n.buildTime * 0.7))
    54→      map.set(n.id, { ...n, x, y: BAND_Y[band], r })
    55→    })
    56→  }
    57→  return map
    58→}
    59→
    60→function edgePath(a: Positioned, b: Positioned): string {
    61→  const y1 = a.y + a.r
    62→  const y2 = b.y - b.r
    63→  const my = (y1 + y2) / 2
    64→  return `M ${a.x} ${y1} C ${a.x} ${my}, ${b.x} ${my}, ${b.x} ${y2}`
    65→}

    67→function nodeFill(n: GraphNode): { fill: string; fillOpacity: number } {
    68→  if (n.kind === 'proc-macro') return { fill: 'var(--chart-5)', fillOpacity: 0.7 }
    69→  if (n.kind === 'external') return { fill: 'oklch(0.30 0.005 60)', fillOpacity: 1 }
    70→  if (n.critical) return { fill: 'var(--chart-1)', fillOpacity: 0.85 }
    71→  return { fill: 'oklch(0.35 0.03 50)', fillOpacity: 1 }
    72→}

   264→                    cy={p.y}

   288→                  </text>
   289→                </g>

   291→            })}
   292→          </g>
   293→        </svg>

   295→        {/* hover tooltip */}
   296→        {hoveredNode && (
   297→          <div
   298→            role="tooltip"
