    72→function ImpactTile({
    73→  label,
    74→  value,
    75→  decimals = 0,
    76→  prefix = '',
    77→  suffix = '',
    78→  cost = false,
    79→}: {
    80→  label: string
    81→  value: number
    82→  decimals?: number
    83→  prefix?: string
    84→  suffix?: string
    85→  cost?: boolean
    86→}) {
    87→  // negative deltas are genuine improvements (e.g. de-dup wins in upgrade-dep)
    88→  const improvement = value < 0
    89→  return (
    90→    <div
    91→      className={cn(
    92→        'rounded-lg border px-3 py-2.5 transition-shadow',
    93→        improvement
    94→          ? 'border-emerald-500/30 bg-emerald-500/[0.07] shadow-[0_0_16px_-8px_oklch(0.72_0.17_160/60%)]'
    95→          : 'border-border/70 bg-muted/20',
    96→      )}
    97→    >
    98→      <p className="text-[10px] uppercase tracking-wide text-muted-foreground">{label}</p>
    99→      <p
   100→        className={cn(
   101→          'mt-1 font-mono text-lg font-semibold',
   102→          improvement ? 'text-emerald-300' : cost ? 'text-red-300' : 'text-foreground',
   103→        )}
   104→      >
   105→        {improvement && <TrendingDown className="mr-1 inline size-3.5 align-baseline" aria-hidden />}
   106→        <CountUp
   107→          value={Math.abs(value)}
   108→          decimals={decimals}
   109→          prefix={improvement ? '−' : prefix}
   110→          suffix={suffix}
   111→        />
   112→      </p>
   113→    </div>
   114→  )

   183→function AddDepReport({ query }: { query: QueryLike<ImpactPayload> }) {
   184→  const [pending, setPending] = useState<string | null>(null)
   185→  const { toast } = useToast()
   186→  const activeWs = useWorkspaceStore((s) => s.active)
   187→  const enqueueDiff = useDiffQueueStore((s) => s.enqueue)
   188→  const d = query.data && query.data.kind === 'add-dep' ? (query.data as AddDepImpact) : undefined
   189→
   190→  if (query.isPending || (query.data === undefined && !query.isError)) {
   191→    return (
   192→      <Panel title="Dependency impact">
   193→        <TilesSkeleton />
   194→      </Panel>
   195→    )
   196→  }
   197→
   198→  if (!d) {
   199→    return (
   200→      <Panel title="Dependency impact">
   201→        <QueryError message={query.error?.message ?? 'unknown error'} onRetry={() => query.refetch()} />
   202→      </Panel>
   203→    )
   204→  }
   205→
   206→  const tiles: {
   207→    label: string
   208→    value: number
   209→    decimals?: number
   210→    prefix?: string
   211→    suffix?: string
   212→    cost?: boolean
   213→  }[] = [
   214→    { label: 'crates added', value: d.cratesAdded },
   215→    { label: 'proc-macros added', value: d.procMacrosAdded },
   216→    { label: 'target size', value: d.targetMB, decimals: 1, suffix: ' MB' },
   217→    { label: 'transitive deps', value: d.transitiveDeps },
   218→    { label: 'clean build', value: d.cleanDelta, decimals: 1, prefix: '+', suffix: 's', cost: true },
   219→    { label: 'incremental', value: d.incrementalDelta, decimals: 1, prefix: '+', suffix: 's', cost: true },
   220→    { label: 'CI', value: d.ciDelta, prefix: '+', suffix: 's', cost: true },
   221→  ]
   222→
   223→  return (
   224→    <Panel
   225→      title={`Dependency impact — ${d.crate} ${d.version}`}
   226→      subtitle="what enters your tree if you add this crate"
   227→      actions={<MeasurementBadge status={d.measurementStatus} />}
   228→    >
   229→      <div className="space-y-5">
   230→        <div className="grid grid-cols-2 gap-2.5 md:grid-cols-4">
   231→          {tiles.map((t) => (
   232→            <ImpactTile key={t.label} {...t} />
   233→          ))}
   234→        </div>
   235→
   236→        <section>
   237→          <SectionLabel>Cascading rebuild chain</SectionLabel>
   238→          <div className="mt-3">
   239→            <ChainStepper items={d.chain} />
   240→          </div>
   241→        </section>
   242→
   243→        <section>
   244→          <SectionLabel>Security &amp; supply chain</SectionLabel>
   245→          <ul className="mt-2.5 space-y-1.5">
   246→            {d.security.map((s, i) => (
   247→              <li key={i} className="flex items-start gap-2 text-xs">
   248→                {s.level === 'warn' ? (
   249→                  <ShieldAlert className="mt-0.5 size-3.5 shrink-0 text-amber-400" />
   250→                ) : (
   251→                  <ShieldCheck className="mt-0.5 size-3.5 shrink-0 text-teal-300" />
   252→                )}
   253→                <span className={s.level === 'warn' ? 'text-amber-200/90' : 'text-muted-foreground'}>
   254→                  {s.label}
   255→                </span>
   256→              </li>
   257→            ))}
   258→          </ul>
   259→        </section>
   260→
   261→        <section>
   262→          <SectionLabel>Suggestions</SectionLabel>
   263→          <ol className="mt-2.5 space-y-2">
   264→            {d.suggestions.map((s, i) => (
   265→              <li
   266→                key={i}
   267→                className="flex items-center justify-between gap-3 rounded-lg border border-border/60 bg-muted/20 px-3 py-2"
   268→              >
   269→                <p className="text-xs leading-snug">
   270→                  <span className="mr-2 font-mono text-muted-foreground">{String(i + 1).padStart(2, '0')}</span>
   271→                  {s}
   272→                </p>
   273→                <Button
   274→                  size="sm"
   275→                  variant="outline"
   276→                  className="h-7 shrink-0 text-[11px]"
   277→                  onClick={() => setPending(s)}
   278→                >
   279→                  Apply suggestion
   280→                </Button>
   281→              </li>
   282→            ))}
   283→          </ol>
   284→        </section>
   285→      </div>
   286→
   287→      <Dialog open={pending !== null} onOpenChange={(o) => !o && setPending(null)}>
   288→        <DialogContent className="max-w-md">
   289→          <DialogHeader>
   290→            <DialogTitle>Queue a reviewable diff?</DialogTitle>
   291→            <DialogDescription>
   292→              Wanyrix never modifies your repository silently. A reviewable diff will be proposed (Gate 19).
   293→            </DialogDescription>
   294→          </DialogHeader>
   295→          <div className="rounded-lg border border-border/70 bg-muted/30 px-3 py-2.5 font-mono text-[11px] leading-relaxed text-foreground/90">
   296→            {pending}
   297→          </div>
   298→          <div className="flex justify-end gap-2">
   299→            <Button size="sm" variant="ghost" onClick={() => setPending(null)}>
   300→              Cancel
   301→            </Button>
   302→            <Button
   303→              size="sm"
   304→              onClick={() => {
   305→                if (!pending) return
   306→                enqueueDiff({
   307→                  id: `diff-${Date.now()}`,
   308→                  workspace: activeWs,
   309→                  at: Date.now(),
   310→                  source: 'simulator:add-dep',
   311→                  kind: 'config',
   312→                  title: `Dependency change — ${d.crate} ${d.version}`,
   313→                  target: `${d.crate} ${d.version} · Cargo.toml`,
   314→                  suggestion: pending,
   315→                  estimate: `+${d.cleanDelta}s clean · +${d.ciDelta}s CI per build`,
   316→                  status: 'pending',
   317→                })
   318→                toast({
   319→                  title: 'Diff queued for review',
   320→                  description: 'Open Pending diffs in the topbar to inspect or export it (Gate 19).',
   321→                })
   322→                setPending(null)
   323→              }}
   324→            >
   325→              Queue as diff
   326→            </Button>
   327→          </div>
   328→        </DialogContent>
   329→      </Dialog>
   330→    </Panel>
   331→  )
   332→}
