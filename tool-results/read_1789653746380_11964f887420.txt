   162→function AddDepReport({ query }: { query: QueryLike<ImpactPayload> }) {
   163→  const [pending, setPending] = useState<string | null>(null)
   164→  const { toast } = useToast()
   165→  const activeWs = useWorkspaceStore((s) => s.active)
   166→  const enqueueDiff = useDiffQueueStore((s) => s.enqueue)
   167→  const d = query.data && query.data.kind === 'add-dep' ? (query.data as AddDepImpact) : undefined
   168→
   169→  if (query.isPending || (query.data === undefined && !query.isError)) {
   170→    return (
   171→      <Panel title="Dependency impact">
   172→        <TilesSkeleton />
   173→      </Panel>
   174→    )
   175→  }
   176→
   177→  if (!d) {
   178→    return (
   179→      <Panel title="Dependency impact">
   180→        <QueryError message={query.error?.message ?? 'unknown error'} onRetry={() => query.refetch()} />
   181→      </Panel>
   182→    )
   183→  }
   184→
   185→  const tiles: {
   186→    label: string
   187→    value: number
   188→    decimals?: number
   189→    prefix?: string
   190→    suffix?: string
   191→    cost?: boolean
   192→  }[] = [
   193→    { label: 'crates added', value: d.cratesAdded },
   194→    { label: 'proc-macros added', value: d.procMacrosAdded },
   195→    { label: 'target size', value: d.targetMB, decimals: 1, suffix: ' MB' },
   196→    { label: 'transitive deps', value: d.transitiveDeps },
   197→    { label: 'clean build', value: d.cleanDelta, decimals: 1, prefix: '+', suffix: 's', cost: true },
   198→    { label: 'incremental', value: d.incrementalDelta, decimals: 1, prefix: '+', suffix: 's', cost: true },
   199→    { label: 'CI', value: d.ciDelta, prefix: '+', suffix: 's', cost: true },
   200→  ]
   201→
   202→  return (
   203→    <Panel
   204→      title={`Dependency impact — ${d.crate} ${d.version}`}
   205→      subtitle="what enters your tree if you add this crate"
   206→      actions={<MeasurementBadge status={d.measurementStatus} />}
   207→    >
   208→      <div className="space-y-5">
   209→        <div className="grid grid-cols-2 gap-2.5 md:grid-cols-4">
   210→          {tiles.map((t) => (
   211→            <ImpactTile key={t.label} {...t} />
   212→          ))}
   213→        </div>
   214→
   215→        <section>
   216→          <SectionLabel>Cascading rebuild chain</SectionLabel>
   217→          <div className="mt-3">
   218→            <ChainStepper items={d.chain} />
   219→          </div>
   220→        </section>
   221→
   222→        <section>
   223→          <SectionLabel>Security &amp; supply chain</SectionLabel>
   224→          <ul className="mt-2.5 space-y-1.5">
   225→            {d.security.map((s, i) => (
   226→              <li key={i} className="flex items-start gap-2 text-xs">
   227→                {s.level === 'warn' ? (
   228→                  <ShieldAlert className="mt-0.5 size-3.5 shrink-0 text-amber-400" />
   229→                ) : (
   230→                  <ShieldCheck className="mt-0.5 size-3.5 shrink-0 text-teal-300" />
   231→                )}
   232→                <span className={s.level === 'warn' ? 'text-amber-200/90' : 'text-muted-foreground'}>
   233→                  {s.label}
   234→                </span>
   235→              </li>
   236→            ))}
   237→          </ul>
   238→        </section>
   239→
   240→        <section>
   241→          <SectionLabel>Suggestions</SectionLabel>
   242→          <ol className="mt-2.5 space-y-2">
   243→            {d.suggestions.map((s, i) => (
   244→              <li
   245→                key={i}
   246→                className="flex items-center justify-between gap-3 rounded-lg border border-border/60 bg-muted/20 px-3 py-2"
   247→              >
   248→                <p className="text-xs leading-snug">
   249→                  <span className="mr-2 font-mono text-muted-foreground">{String(i + 1).padStart(2, '0')}</span>
   250→                  {s}
   251→                </p>
   252→                <Button
   253→                  size="sm"
   254→                  variant="outline"
   255→                  className="h-7 shrink-0 text-[11px]"
   256→                  onClick={() => setPending(s)}
   257→                >
   258→                  Apply suggestion
   259→                </Button>
   260→              </li>
   261→            ))}
   262→          </ol>
   263→        </section>
   264→      </div>
   265→
   266→      <Dialog open={pending !== null} onOpenChange={(o) => !o && setPending(null)}>
   267→        <DialogContent className="max-w-md">
   268→          <DialogHeader>
   269→            <DialogTitle>Queue a reviewable diff?</DialogTitle>
   270→            <DialogDescription>
   271→              Ferrix never modifies your repository silently. A reviewable diff will be proposed (Gate 19).
   272→            </DialogDescription>
   273→          </DialogHeader>
   274→          <div className="rounded-lg border border-border/70 bg-muted/30 px-3 py-2.5 font-mono text-[11px] leading-relaxed text-foreground/90">
   275→            {pending}
   276→          </div>
   277→          <div className="flex justify-end gap-2">
   278→            <Button size="sm" variant="ghost" onClick={() => setPending(null)}>
   279→              Cancel
   280→            </Button>
   281→            <Button
   282→              size="sm"
   283→              onClick={() => {
   284→                if (!pending) return
   285→                enqueueDiff({
   286→                  id: `diff-${Date.now()}`,
   287→                  workspace: activeWs,
   288→                  at: Date.now(),
   289→                  source: 'simulator:add-dep',
   290→                  kind: 'config',
   291→                  title: `Dependency change — ${d.crate} ${d.version}`,
   292→                  target: `${d.crate} ${d.version} · Cargo.toml`,
   293→                  suggestion: pending,
   294→                  estimate: `+${d.cleanDelta}s clean · +${d.ciDelta}s CI per build`,
   295→                  status: 'pending',
   296→                })
   297→                toast({
   298→                  title: 'Diff queued for review',
   299→                  description: 'Open Pending diffs in the topbar to inspect or export it (Gate 19).',
   300→                })
   301→                setPending(null)
   302→              }}
   303→            >
   304→              Queue as diff
   305→            </Button>
   306→          </div>
   307→        </DialogContent>
   308→      </Dialog>
   309→    </Panel>
   310→  )
   311→}

   429→function SplitReport({

   437→  const [done, setDone] = useState<boolean[]>([])
   438→  const d = query.data && query.data.kind === 'split-crate' ? (query.data as SplitImpact) : undefined
   439→
   440→  if (query.isPending || (query.data === undefined && !query.isError)) {
   441→    return (
   442→      <Panel title="Architecture proposal">
   443→        <TilesSkeleton count={4} />

   445→    )
   446→  }
   447→
   448→  if (!d) {
   449→    return (
   450→      <Panel title="Architecture proposal">
   451→        <QueryError message={query.error?.message ?? 'unknown error'} onRetry={() => query.refetch()} />
   452→      </Panel>
   453→    )
   454→  }
   455→

   460→      toast({ title: 'All steps marked — create the experiment to verify' })

   468→      title={`Architecture proposal — split ${d.source}`}

   482→            <div className="mt-3 flex flex-wrap gap-1.5">
